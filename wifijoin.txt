echo '### wifijoin: join the WiFi network in WIFI.TXT with the internal RTL8188FTV ###'
setenv a 0x80008000
usb start
fatload usb 0 ${a} rtljoin.bin
mw.b 83d00000 0 400
fatload usb 0 83d00000 WIFI.TXT
fatload usb 0 83c00000 usbfast.bin
go 83c00000
fatload usb 0 82000000 rtl8188fufw.bin
usb port 1
usb reset
go ${a} 82000000 ${filesize} 83d00000
